\section{Publications}

\subsection{Conference and workshop papers}
\begin{pubs}
  \item \me. ``Curated to Win: Certifying Reward-Model and Judge Selection Against Evaluation-Set Curation.'' \textit{First Workshop on Autonomous Research and Recursive Self-Improvement (ARRSI), AACL-IJCNLP 2026}, Nov 2026. \textbf{Accepted.} Sole author.\enspace\lnk{https://adrianerlikhman.is-a.dev/papers/curated-to-win-arrsi2026.pdf}{PDF}\enspace\lnk{https://github.com/adrian-erlikhman/curated-to-win}{Code}
  \begin{itemize}
    \item Proposed a curation certificate for choosing a reward model or LLM judge by validation accuracy: how much of the set could be removed before the winner changes. In benchmark weight it is at least the top-two gap; counted in items it is exact.
    \item Removing 12 particular items of RewardBench's 2,985 makes the runner-up the winner (11 under its own scoring code). On 200-item samples, picks with a top-two gap above 2 points match the full-benchmark winner 87\% of the time (43\% otherwise).
  \end{itemize}
  \item \me, M.~Tarekegn, P.~Juang. ``Interpretable Multilingual Attribution of Frontier LLMs.'' \textit{IEEE BigData 2026 High School Symposium}, Phoenix, AZ, Dec 2026. \textbf{Accepted; Best Paper Finalist.} Co-first author.\enspace\lnk{https://adrianerlikhman.is-a.dev/papers/langllm-bigdata2026.pdf}{PDF}
  \begin{itemize}
    \item 840 essays from five frontier models in seven languages, English to Hindi, attributed with 21 interpretable features from Universal Dependencies parses: 59--72\% accuracy in every language against a 20\% chance floor, with no decline detected as a language's resources fall.
    \item The features transfer between languages better than character n-grams (49\% against 41\%) and hold at 57--70\% on machine-translated text; across 4,195 judge calls, four of the five models attribute authorship at chance.
  \end{itemize}
\end{pubs}

\subsection{Journal articles}
\begin{pubs}[resume]
  \item \me, R.~Erlikhman. ``Does Portfolio Optimization Beat Splitting Evenly?'' \textit{Oxford Journal of Student Scholarship} 12(7):510--529, Oct 2026. \textbf{Published.} Co-first author.\enspace\lnk{https://adrianerlikhman.is-a.dev/papers/portfolio-optimization-vs-equal-weight.pdf}{PDF}
  \begin{itemize}
    \item Equal weights beat Markowitz in 74\% of 250 calibrated simulations (Sharpe 0.70 against 0.51). The loss traces to estimating expected returns, and shrinkage closes it only with roughly two decades of stationary data.
  \end{itemize}
  \item \me, R.~Erlikhman. ``How Robust Are the Legatum Prosperity Index Rankings? A Monte Carlo and Sensitivity Analysis of Weighting, Normalization, and Aggregation.'' \textit{Journal of High School Science}. \textbf{In revision.} Co-first author.\enspace\lnk{https://adrianerlikhman.is-a.dev/papers/legatum-robustness-audit.pdf}{PDF}
  \begin{itemize}
    \item Reproduced every published score for 167 countries, then ran 10,000 draws under each of three weight priors and a 131,072-run Sobol' design over weighting, normalization and aggregation.
    \item Derived rank leverage, c\sq(1\,\textminus\,\rhosym\sq), which tracks Sobol' total effects (Spearman 0.91--0.99) at a fraction of the compute. The ranking holds as a whole (median Spearman \geqsym\ 0.9899) but not mid-table, where the median 90\% rank interval spans 36 places.
  \end{itemize}
\end{pubs}

\subsection{Preprints}
\begin{pubs}[resume]
  \item \me, M.~Tarekegn, P.~Juang. ``Whose Line Is It Anyway? Auditing Self- and Other-Recognition in LLM Judges.'' Preprint, arXiv forthcoming. Co-first author.\enspace\lnk{https://github.com/adrian-erlikhman/self-recognition-peer-baseline}{Code and data}
  \begin{itemize}
    \item Introduced a peer baseline that scores each LLM judge's self-attribution against how often the other judges name it on the same text, separating self-recognition from text any model can identify.
    \item Nine judges, 1,270 texts, over 19,000 judgments: the usual test flags 15 of 37 cases, 8 survive a false-alarm control, 3 beat the pooled peers and none beats its strongest single peer. An 18-feature stylometric classifier finds the true author 86.3\% of the time, above every judge.
  \end{itemize}
\end{pubs}
