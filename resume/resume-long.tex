\section{Research}

\paper{Self-Attribution Is Not Self-Recognition: A Peer Baseline for LLM Judges}{}%
  {First author\sep with R.\ Erlikhman, M.\ Tarekegn\sep advised by Philo Juang (Google DeepMind)}%
  {Under review, JUDGe @ NeurIPS 2026}
\begin{itemize}
  \item Introduced a peer baseline that scores each LLM judge's self-attribution rate against what the other judges get on the same text, separating self-recognition from text that is simply easy to identify.
  \item Five frontier judges over 190 responses: Claude names its own writing 86.8\% of the time against 53.9\% for its peers (+32.9 points, Fisher \textit{p}\,=\,1.6\,\texttimes\,10\textsuperscript{\textminus4}), and an 18-feature stylometric classifier (86.3\%) beats every judge.
\end{itemize}

\paper{Do LLM Fingerprints Survive Outside English?}{https://github.com/adrian-erlikhman/LangLLM}%
  {First author\sep with M.\ Tarekegn, R.\ Erlikhman\sep advised by Philo Juang (Google DeepMind)}%
  {Poster submitted, IEEE MIT URTC 2026}
\begin{itemize}
  \item 840 responses from five frontier models in seven languages, English to Hindi, attributed with 21 Universal Dependencies features: 59--72\% accuracy in every language (chance is 20\%), no drop for low-resource languages, and it survives machine translation.
  \item Across 4,195 judge calls, four of the five models attribute authorship at chance, while Claude's self-recognition holds even in Hindi.\enspace\lnk{https://adrianerlikhman.is-a.dev/papers/langllm-abstract-urtc2026.pdf}{abstract}\enspace\lnk{https://adrianerlikhman.is-a.dev/papers/langllm-poster-urtc2026.pdf}{poster}\enspace\lnk{https://github.com/adrian-erlikhman/LangLLM}{code}
\end{itemize}

\paper{Proximity, Not Transmission: What the Patent Record Can and Cannot Show About NLP and Surveillance}{https://github.com/adrian-erlikhman/earshot}%
  {First author\sep with Michael Tarekegn}%
  {Submitting, AI for Peace @ NeurIPS 2026}
\begin{itemize}
  \item Linked 127,851 ACL Anthology papers to the 9,048 USPTO patents that cite them, and coded what each patent is for with a rubric that separates a system's operator from its subject; revealing the applicant flips 10 of 31 non-neutral labels.
  \item Name-blind coding of 2,374 patents finds 0.46\% surveillance, no standout subfield, no trend and no detectable difference from science-citing patents of the same era (Mantel--Haenszel OR 1.03, 95\% CI 0.38--2.78).\enspace\lnk{https://adrianerlikhman.is-a.dev/papers/earshot-abstract-neurips2026.pdf}{abstract}\enspace\lnk{https://github.com/adrian-erlikhman/earshot}{code}
\end{itemize}

\paper{How Robust Are the Legatum Prosperity Index Rankings? A Monte Carlo and Sensitivity Analysis of Weighting, Normalization, and Aggregation}{https://adrianerlikhman.is-a.dev/papers/legatum-robustness-audit.pdf}%
  {First author\sep with Ryan Erlikhman}%
  {Accepted with revisions, Journal of High School Science}
\begin{itemize}
  \item Reproduced every published score for 167 countries exactly, then varied weights, normalization and aggregation: 10,000 draws under each of three weight priors and a 131,072-run Sobol' design.
  \item Derived rank leverage, c\sq(1\,\textminus\,\rhosym\sq), the only part of a weight's influence that can reorder countries; it tracks Sobol' total effects (Spearman 0.91--0.99). The ranking holds as a whole (median Spearman \geqsym\ 0.9899) but not mid-table, where the median 90\% rank interval spans 36 places.\enspace\lnk{https://adrianerlikhman.is-a.dev/papers/legatum-robustness-audit.pdf}{pdf}
\end{itemize}

\paper{Does Portfolio Optimization Beat Splitting Evenly? Markowitz vs.\ the 1/N Portfolio}{https://adrianerlikhman.is-a.dev/papers/portfolio-optimization-vs-equal-weight.pdf}%
  {With Ryan Erlikhman}%
  {Under review, Oxford Journal of Student Scholarship}
\begin{itemize}
  \item Equal weights beat Markowitz in 74\% of 250 calibrated simulations (Sharpe 0.70 vs.\ 0.51). The loss traces to estimating expected returns, which shrinkage fixes only with about two decades of stationary data.\enspace\lnk{https://adrianerlikhman.is-a.dev/papers/portfolio-optimization-vs-equal-weight.pdf}{pdf}
\end{itemize}
