\section{Research}

\paper{Self-Attribution Is Not Self-Recognition: A Peer Baseline for LLM Judges}{}%
  {Co-first author, with R.\ Erlikhman, M.\ Tarekegn; advised by Philo Juang (Google DeepMind)}%
  {Under review, JUDGe @ NeurIPS 2026}
\begin{itemize}
  \item Introduced a peer baseline that scores each LLM judge's self-attribution rate against what the other four judges get on the same text, separating genuine self-recognition from text that is simply easy to identify -- the confound prior self-recognition results leave in.
  \item Five frontier judges over the same 190 responses under two elicitation formats: Claude names its own writing 86.8\% of the time against 53.9\% for its peers (+32.9 points, Fisher \textit{p}\,=\,1.6\,\texttimes\,10\textsuperscript{\textminus4}), GPT shows the same edge, two models are neutral and Grok lands below the chance floor on itself.
  \item An 18-feature interpretable stylometric classifier recovers the true author 86.3\% of the time, above every judge, so the signal is in the text and most judges simply do not use it. Reproducible in one command, with no API key.
\end{itemize}

\paper{Do LLM Fingerprints Survive Outside English?}{https://github.com/adrian-erlikhman/LangLLM}%
  {Co-first author}%
  {Under review, IEEE BigData 2026 HS Symposium}
\begin{itemize}
  \item Collected 840 essays from five frontier models across seven languages, English to Hindi, and attributed them with 21 interpretable features computed from Universal Dependencies parses: 59--72\% accuracy in every language against a 20\% chance floor, with no decline detected as a language's resources fall.
  \item Character n-grams are more accurate within a language, but the interpretable features transfer between languages better (49\% against 41\%), and attribution still holds at 57--70\% on machine-translated text.
  \item Across 4,195 judge calls, four of the five models attribute authorship at chance (20--25\%) while Claude's self-recognition holds even in Hindi.\enspace\lnk{https://adrianerlikhman.is-a.dev/papers/langllm-abstract-urtc2026.pdf}{abstract}\enspace\lnk{https://adrianerlikhman.is-a.dev/papers/langllm-poster-urtc2026.pdf}{poster}\enspace\lnk{https://github.com/adrian-erlikhman/LangLLM}{code}
\end{itemize}

\paper{Proximity, Not Transmission: What the Patent Record Can and Cannot Show About NLP and Surveillance}{https://github.com/adrian-erlikhman/earshot}%
  {Co-first author, with Michael Tarekegn}%
  {Submitting, AI for Peace @ NeurIPS 2026}
\begin{itemize}
  \item Linked 127,851 ACL Anthology papers to the 9,048 USPTO patents that cite them, then coded what each patent is actually for with a rubric that separates a system's operator from its subject; revealing the applicant flips 10 of 31 non-neutral labels, which is why the coding is name-blind.
  \item Name-blind coding of 2,374 patents finds 0.46\% surveillance, no standout subfield, no trend over time, and no detectable difference from science-citing patents of the same era (Mantel--Haenszel OR 1.03, 95\% CI 0.38--2.78). The honest finding is a null one, and the paper says so.\enspace\lnk{https://adrianerlikhman.is-a.dev/papers/earshot-abstract-neurips2026.pdf}{abstract}\enspace\lnk{https://github.com/adrian-erlikhman/earshot}{code}
\end{itemize}

\paper{How Robust Are the Legatum Prosperity Index Rankings?\newline A Monte Carlo and Sensitivity Analysis of Weighting, Normalization, and Aggregation}{https://adrianerlikhman.is-a.dev/papers/legatum-robustness-audit.pdf}%
  {Co-first author, with Ryan Erlikhman}%
  {Accepted with revisions, Journal of High School Science}
\begin{itemize}
  \item Reproduced every published score for 167 countries exactly, then varied the choices the index hides: 10,000 draws under each of three weight priors, and a 131,072-run Sobol' design over weighting, normalization and aggregation.
  \item Derived rank leverage, c\sq(1\,\textminus\,\rhosym\sq) -- the only part of a weight's influence that can actually reorder countries -- and showed it tracks Sobol' total effects (Spearman 0.91--0.99) at a fraction of the compute. The ranking holds as a whole (median Spearman \geqsym\ 0.9899) but not mid-table, where the median 90\% rank interval spans 36 places.\enspace\lnk{https://adrianerlikhman.is-a.dev/papers/legatum-robustness-audit.pdf}{pdf}
\end{itemize}

\paper{Does Portfolio Optimization Beat Splitting Evenly? Markowitz vs.\ the 1/N Portfolio}{https://adrianerlikhman.is-a.dev/papers/portfolio-optimization-vs-equal-weight.pdf}%
  {With Ryan Erlikhman}%
  {Accepted with revisions, Oxford Journal of Student Scholarship}
\begin{itemize}
  \item Equal weights beat Markowitz in 74\% of 250 calibrated simulations (Sharpe 0.70 against 0.51). The loss traces to estimating expected returns, and shrinkage closes it only with roughly two decades of stationary data -- more history than most investors have.\enspace\lnk{https://adrianerlikhman.is-a.dev/papers/portfolio-optimization-vs-equal-weight.pdf}{pdf}
\end{itemize}
