\section{Research}

\paper{Whose Line Is It Anyway? Auditing Self- and Other-Recognition in LLM Judges}{https://github.com/adrian-erlikhman/self-recognition-peer-baseline}{Preprint, arXiv forthcoming}
\begin{itemize}
  \item \who{Co-first author, advised by Philo Juang (Google DeepMind)}Nine LLM judges, 1,270 texts: the usual self-recognition test flags 15 of 37 cases, 3 beat a peer baseline and none beats its strongest single peer; an 18-feature stylometric classifier (86.3\%) beats every judge.
\end{itemize}

\paper{Do LLM Fingerprints Survive Outside English?}{https://adrianerlikhman.is-a.dev/papers/langllm-bigdata2026.pdf}{Best Paper Finalist, IEEE BigData 2026 HS Symposium}
\begin{itemize}
  \item \who{Co-first author}840 essays from five models in seven languages: attribution holds at 59--72\% against 20\% chance, with no detected drop for low-resource languages, and survives Google Translate.
\end{itemize}

\paper{Curated to Win: Certifying Reward-Model and Judge Selection Against Evaluation-Set Curation}{https://adrianerlikhman.is-a.dev/papers/curated-to-win-arrsi2026.pdf}{Accepted, ARRSI Workshop @ AACL-IJCNLP 2026}
\begin{itemize}
  \item \who{Sole author}A certificate for how much of a validation set can be removed before the chosen reward model or LLM judge changes: on RewardBench, 12 of 2,985 items.
\end{itemize}

\paper{How Robust Are the Legatum Prosperity Index Rankings?}{https://adrianerlikhman.is-a.dev/papers/legatum-robustness-audit.pdf}{Accepted with revisions, Journal of High School Science}
\begin{itemize}
  \item \who{Co-first author, with Ryan Erlikhman}Derived rank leverage, c\sq(1\,\textminus\,\rhosym\sq), which tracks Sobol' total effects (Spearman 0.91--0.99); the ranking holds globally but not mid-table, where the median 90\% rank interval spans 36 places.
\end{itemize}
