\section{Research}

\paper{Interpretable Multilingual Attribution of Frontier LLMs}{}{Best Paper Finalist, IEEE BigData 2026 HS Symposium\sep Co-first author, with Michael Tarekegn and Philo Juang}
\begin{itemize}
  \item Attributed 840 essays from five frontier models in seven languages at 59--72\% accuracy (20\% chance), with no detected drop in low-resource languages; the signal survives Google Translate.
\end{itemize}

\paper{Curated to Win: Certifying Reward-Model and Judge Selection Against Evaluation-Set Curation}{}{Accepted, ARRSI Workshop at AACL-IJCNLP 2026\sep Sole author}
\begin{itemize}
  \item Introduced a certificate for how much of a validation set can be removed before the chosen reward model or LLM judge changes; on RewardBench, removing 12 of 2,985 items flips the winner.
\end{itemize}

\paper{Who Wrote This? LLM Judges Encode Authorship Without a Detectable Sense of Self}{}{Accepted, WLLFM Workshop at IEEE BigData 2026\sep Co-first author, with Michael Tarekegn and Philo Juang}
\begin{itemize}
  \item Showed that probes on four open-weight LLM judges' hidden states identify the author of 960 texts (AUC 0.90--0.98), yet no judge's own states beat its strongest peer's.
\end{itemize}

\paper{Does Portfolio Optimization Beat Splitting Evenly?}{}{Published, Oxford Journal of Student Scholarship\sep Co-first author, with Ryan Erlikhman}
\begin{itemize}
  \item Showed that equal weights beat Markowitz in 74\% of 250 calibrated simulations (Sharpe 0.70 vs.\ 0.51), and traced the gap to estimating expected returns.
\end{itemize}

\paper{Whose Line Is It Anyway? Auditing Self- and Other-Recognition in LLM Judges}{}{Preprint, arXiv forthcoming\sep Co-first author, with Michael Tarekegn and Philo Juang}
\begin{itemize}
  \item Tested nine LLM judges on 1,270 texts against a peer baseline: the usual self-recognition test flags 15 of 37 cases, but none beats its strongest single peer, and an 18-feature stylometric classifier beats every judge.
\end{itemize}

\paper{How Robust Are the Legatum Prosperity Index Rankings?}{}{In revision, Journal of High School Science\sep Co-first author, with Ryan Erlikhman}
\begin{itemize}
  \item Derived rank leverage, c\sq(1\,\textminus\,\rhosym\sq), which tracks Sobol' total effects (Spearman 0.91--0.99); the ranking holds overall but not mid-table, where the median 90\% rank interval spans 36 places.
\end{itemize}
