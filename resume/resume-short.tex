\section{Research}

\paper{Self-Attribution Is Not Self-Recognition: A Peer Baseline for LLM Judges}{}{Under review, JUDGe @ NeurIPS 2026}
\begin{itemize}
  \item \who{First author, advised by Philo Juang (Google DeepMind)}Five frontier LLM judges over 190 responses, each scored against a peer baseline: Claude names its own writing 86.8\% against 53.9\% for its peers (\textit{p}\,=\,1.6\,\texttimes\,10\textsuperscript{\textminus4}), and an 18-feature stylometric classifier (86.3\%) beats every judge.
\end{itemize}

\paper{Do LLM Fingerprints Survive Outside English?}{https://github.com/adrian-erlikhman/LangLLM}{Under review, IEEE BigData 2026 HS Symposium}
\begin{itemize}
  \item \who{First author, advised by Philo Juang (Google DeepMind)}840 essays from five models in seven languages: attribution holds at 59--72\% against 20\% chance, with no detected drop for low-resource languages, and survives Google Translate.\enspace\lnk{https://adrianerlikhman.is-a.dev/papers/langllm-abstract-urtc2026.pdf}{abstract}\enspace\lnk{https://adrianerlikhman.is-a.dev/papers/langllm-poster-urtc2026.pdf}{poster}\enspace\lnk{https://github.com/adrian-erlikhman/LangLLM}{code}
\end{itemize}

\paper{Proximity, Not Transmission:\newline What the Patent Record Can and Cannot Show About NLP and Surveillance}{https://github.com/adrian-erlikhman/earshot}{Submitting, AI for Peace @ NeurIPS 2026}
\begin{itemize}
  \item \who{First author, with Michael Tarekegn}Linked 127,851 ACL Anthology papers to the 9,048 USPTO patents citing them; name-blind coding finds 0.46\% surveillance, no different from science-citing patents of the same era (OR 1.03, 95\% CI 0.38--2.78).\enspace\lnk{https://adrianerlikhman.is-a.dev/papers/earshot-abstract-neurips2026.pdf}{abstract}\enspace\lnk{https://github.com/adrian-erlikhman/earshot}{code}
\end{itemize}

\paper{How Robust Are the Legatum Prosperity Index Rankings?}{https://adrianerlikhman.is-a.dev/papers/legatum-robustness-audit.pdf}{Accepted with revisions, Journal of High School Science}
\begin{itemize}
  \item \who{First author, with Ryan Erlikhman}Derived rank leverage, c\sq(1\,\textminus\,\rhosym\sq), which tracks Sobol' total effects over a 131,072-run design (Spearman 0.91--0.99); the ranking holds globally but not mid-table, where the median 90\% rank interval spans 36 places.\enspace\lnk{https://adrianerlikhman.is-a.dev/papers/legatum-robustness-audit.pdf}{pdf}
\end{itemize}
